# Le Lis et le Lion — Royaume de Castille (k_castille)
# En vanilla, le duché de Tolède (d_toledo, qui contient c_madrid) est de_jure sous
# k_andalusia, alors qu'historiquement Tolède est reconquise et intégrée à la Castille dès
# 1085. On corrige ce rattachement de_jure, ce qui permet ensuite de fixer Madrid comme
# capitale de_jure du royaume de Castille (set_capital_county exige que le comté visé soit
# déjà de_jure partie du titre concerné).

d_toledo = {
	1337.5.24 = {
		de_jure_liege = k_castille
	}
}

k_castille = {
	1337.5.24 = {
		effect = {
			set_capital_county = title:c_madrid
		}
	}
}
